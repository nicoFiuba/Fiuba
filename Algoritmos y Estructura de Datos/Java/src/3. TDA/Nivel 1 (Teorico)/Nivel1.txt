public class Rectangulo {

    private double base;
    private double altura;

    public Rectangulo (double base, double altura) {

        if (base > 0) {
            this.base = base;
        } else {
            this.base = 1;
        }

        if (altura > 0) {
            this.altura = altura;
        } else {
            this.altura = 1;
        }
        
    }

    public double getBase() {
        return this.base;
    }

    public double getAltura() {
        return this.altura;
    }

    public void setBase(double nuevaBase) {

        if (nuevaBase > 0) {
            this.base = nuevaBase;
        }

    }

    public void setAltura(double nuevaAltura) {

        if (nuevaAltura > 0) {
            this.altura = nuevaAltura;
        }

    }

    public double getPerimetro() {
        return (base*2 + altura*2);
    }

    public double getArea() {
        return (base*altura);
    }

    @Override
    public String toString() {
        return "La base mide " + this.base + "cm y la altura mide " + this.altura + "cm";
    }
}

public class Fraccion {
    
    private int numerador;
    private int denominador;

    public Fraccion (int numerador, int denominador){

        this.numerador = numerador;

        if (denominador == 0) {
            this.denominador = 1;
        } else {
            this.denominador = denominador;
        }

    }

    public int getNumerador() {
        return this.numerador;
    }

    public int getDenominador() {
        return this.denominador;
    }
    
    private int mcd (int numerador, int denominador) {

        int x = Math.abs(numerador);
        int y = Math.abs(denominador);

        while (y != 0) {
            int valorTemporal = y;
            y = x % y;
            x = valorTemporal;
        }

        return x;
    }

    public void simplificar() {

        int mcd = mcd (this.numerador, this.denominador);
        if (mcd != 0) {
            this.numerador /= mcd;
            this.denominador /= mcd;
        }
        
    }
    
    public Fraccion sumar(Fraccion fraccion2) {

        int nuevoNumerador = (this.numerador * fraccion2.getDenominador()) + (this.denominador * fraccion2.getNumerador());
        int nuevoDenominador = this.denominador * fraccion2.getDenominador();

        Fraccion resultado = new Fraccion (nuevoNumerador, nuevoDenominador);
        resultado.simplificar();

        return resultado;
    }
    
    public Fraccion restar(Fraccion fraccion2) {

        int nuevoNumerador = (this.numerador * fraccion2.getDenominador()) -  (this.denominador * fraccion2.getNumerador());
        int nuevoDenominador = this.denominador * fraccion2.getDenominador();

        Fraccion resultado = new Fraccion (nuevoNumerador, nuevoDenominador);
        resultado.simplificar();

        return resultado;
    }
    
    public Fraccion multiplicar(Fraccion fraccion2) {

        int nuevoNumerador = this.numerador * fraccion2.getNumerador();
        int nuevoDenominador = this.denominador * fraccion2.getDenominador();

        Fraccion resultado = new Fraccion (nuevoNumerador, nuevoDenominador);
        resultado.simplificar();

        return resultado;
    }
    
    public Fraccion dividir(Fraccion fraccion2){

        int nuevoNumerador = this.numerador * fraccion2.getDenominador();
        int nuevoDenominador = this.denominador * fraccion2.getNumerador();

        Fraccion resultado = new Fraccion (nuevoNumerador, nuevoDenominador);
        resultado.simplificar();

        return resultado;
    }

    @Override
    public String toString() {
        return this.numerador + "/" + this.denominador;
    }
}

public class Main {
    
    public static void main (String[] args) {
        
        Fraccion fraccion = new Fraccion(1,2);
        Fraccion fraccion2 = new Fraccion(4, 3);
        Scanner scanner = new Scanner(System.in);

        System.out.println("Que quiere hacer con las fracciones? (+, -, * o /): ");
        char eleccion = scanner.next().toLowerCase.charAt(0);

        Fraccion resultado;

        switch (eleccion) {
            case '+' -> {
                resultado = fraccion.sumar(fraccion2);
                System.out.println("La suma de las fracciones es: " + resultado);
            }
            case '-' -> {
                resultado = fraccion.restar(fraccion2);
                System.out.println("La resta de las fracciones es: " + resultado);
            }
            case '*' -> {
                resultado = fraccion.multiplicar(fraccion2);
                System.out.println("La multiplicacion de las fracciones es: " + resultado);
            }
            case '/' -> {
                resultado = fraccion.dividir(fraccion2);
                System.out.println("La division de las fracciones es: " + resultado);
            }
            default -> System.out.println("Operacion no valida");
        }
        
        scanner.close();
    }
}
