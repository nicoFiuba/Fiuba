public class Escudo {

    private int resitencia;
    private int resistenciaActual;
    private boolean fueDestruido;
    private int contadorDeAtaquesSoportados;

    public Escudo(int resitencia) {

        if (resistencia <= 0) {
            throw new IllegalArgumentException("La resitencia debe ser positiva");
        }

        this.resitencia = resitencia;
        this.resistenciaActual = resistencia;
        this.fueDestruido = false;
        this.contadorDeAtaquesSoportados = 0;

    }

    public boolean estaDestruido() {
        return this.resistenciaActual <= 0;
    }

    public void defenderAtaque(int puntosAtaque) {

        if (puntosAtaque <= 0) {
            return;
        }

        if (this.estaDestruido()) {
            return;
        }

        this.resistenciaActual -= puntosAtaque;
        contadorDeAtaquesSoportados++;

        if (this.estaDestruido()) {
            this.fueDestruido = true;
        }

    }

    public void reparar() {

        if (!this.fueDestruido) {
            this.resistenciaActual = this.resitencia;
        }
    }

    public int contarAtaquesSoportados() {
        return contadorDeAtaquesSoportados;
    }
}

public class AsistenteComercial {

    public PlanComercial elegirPlanMasEconomico(Lista<PlanComercial> planesDisponibles, List<String> serviciosRequeridos) {

        PlanComercial planGanador = null;
        int precioMinimo = Integer.MAX_VALUE;

        planesDisponibles.iniciarCursor();
        while (planesDisponibles.avanzarCursor()) {
            PlanComercial planActual = planesDisponibles.obtenerCursor();

            if (this.elPlanCumpleConLosRequisitos(planActual, serviciosRequeridos)) {

                int precioActual = planActual.obtenerPrecio();

                if (precioActual < precioMinimo) {
                    precioMinimo = precioActual;
                    planGanador = planActual;
                }

            }
        }

        return planGanador;
    }


    private boolean elPlanCumpleConLosRequisitos(PlanComercial plan, List<String> serviciosRequeridos) {

        Lista<Servicio> servicios = plan.obtenerServicios();

        serviciosRequeridos.iniciarCursor();
        while (serviciosRequeridos.avanzarCursor()) {
            String servicioBuscado = serviciosRequeridos.obtenerCursor();

            boolean encontradoYValido = this.buscarYValidarServicio(servicios, servicioBuscado);

            if (!encontradoYValido) {
                return false;
            }

        }

        return true;
    }

    private boolean buscarYValidarServicio(Lista<Servicio> servicios, String servicioBuscado) {

        servicios.iniciarCursor();
        while (servicios.avanzarCursor()) {
            Servicio servicioActual = servicios.obtenerCursor();

            if (servicioActual.obtenerNombre().equals(servicioBuscado)) {
                if (servicioActual.esAdicional()) {
                    return false;
                } else {
                    return true;
                }
            }

        }

        return false;
    }
}
