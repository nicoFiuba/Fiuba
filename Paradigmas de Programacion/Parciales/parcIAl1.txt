    )

Resultado esperado para destino "Roma": 
Map("Buenos Aires" -> 95, "Cordoba" -> 0, "Rosario" -> 0)
(Nota: Mateo gastó 50 hacia Roma, Lucas gastó 45 hacia Roma.
Total BA = 95).

def gastoPorDestino(viajes: List[(String, String, List[(String, String, Int)])], destinoBuscado: String): Map[String, Int] = {
    
    viajes.foldLeft(Map.empty[String, Int]) {
        case(mapaAcumulado, (nombre, ciudadOrigen, vuelos)) => val gastoMochilero = vuelos.foldLeft(0) {
            case(totalVuelos, (origenVuelo, destinoVuelo, precio)) if (destinoVuelo == destinoBuscado) => totalVuelos + precio
            case(totalVuelos, _) => totalVuelos
        }
        
        val gastoAnterior = mapaAcumulado.getOrElse(ciudadOrigen, 0)
        mapaAcumulado + (ciudadOrigen -> (gastoAnterior + gastoMochilero))

        /*
            --- TRAZA DE MEMORIA ("Roma") ---
        
            Estado Inicial: Map()
            Vuelta 1 (Mateo, "Buenos Aires"):
                gastoMochilero = 50 (Solo suma el Madrid -> Roma)
                mapaAcumulado = Map("Buenos Aires" -> 50)

            Vuelta 2 (Sofia, "Cordoba"):
                gastoMochilero = 0 (Ningún vuelo va hacia Roma)
                mapaAcumulado = Map("Buenos Aires" -> 50, "Cordoba" -> 0)

            Vuelta 3 (Lucas, "Buenos Aires"):
                gastoMochilero = 45 (Suma el Manchester -> Roma)
                gastoAnterior = 50 (Lo recupera del Map con getOrElse)
                mapaAcumulado = Map("Buenos Aires" -> 95, "Cordoba" -> 0)

            Vuelta 4 (Ana, "Rosario"):
                gastoMochilero = 0 (Su vuelo sale de Roma hacia Napoles, no cuenta)
                mapaAcumulado = Map("Buenos Aires" -> 95, "Cordoba" -> 0, "Rosario" -> 0)
            
        */
    
    }
}
