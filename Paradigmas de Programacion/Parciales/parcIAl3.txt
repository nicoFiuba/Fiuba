def totalMetricaPorPlataforma(videos: List[(String, String, List[(String, Int)])], metricaBuscada: String): Map[String, Int] = {

    videos.foldLeft(Map.empty[String, Int]) {
        case(mapaAcumulado, (_, plataforma, metricas)) => val totalAcumulado = metricas.foldLeft(0) {
            case(acumulador, (nombreMetrica, cantidad)) if (nombreMetrica == metricaBuscada) => acumulador + cantidad
            case(acumulador, _) => acumulador
        }

        val totalAnterior = mapaAcumulado.getOrElse(plataforma, 0)
        mapaAcumulado + (plataforma -> (totalAnterior + totalAcumulado))

        /*
            --- TRAZA DE MEMORIA ("Likes") ---
            
            Estado Inicial: Map()
            Vuelta 1 (Reel_Linux, "Instagram"):
                totalAcumulado = 1200 (Encuentra la métrica "Likes")
                mapaAcumulado = Map("Instagram" -> 1200)

            Vuelta 2 (TikTok_Scala, "TikTok"):
                totalAcumulado = 4500 (Encuentra la métrica "Likes")
                mapaAcumulado = Map("Instagram" -> 1200, "TikTok" -> 4500)

            Vuelta 3 (Reel_Prolog, "Instagram"):
                totalAcumulado = 500 (Encuentra la métrica "Likes")
                totalAnterior = 1200 (Lo recupera del Map con getOrElse)
                mapaAcumulado = Map("Instagram" -> 1700, "TikTok" -> 4500)

            Vuelta 4 (Short_C, "YouTube"):
                totalAcumulado = 0 (No tiene la métrica "Likes", el fold interno devuelve 0)
                mapaAcumulado = Map("Instagram" -> 1700, "TikTok" -> 4500, "YouTube" -> 0)
                
        */
    
    }
}
