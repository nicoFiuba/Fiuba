% Reglas:
    % Caso Base
    puede_viajar(Origen, Destino, Presupuesto) :-
        viaje(Origen, Destino, Costo),
        Costo =< Presupuesto.

    % Recursion
    puede_viajar(Origen, Destino, Presupuesto) :-
        viaje(Origen, Escala, CostoTramo),
        CostoTramo =< Presupuesto,
        PresupuestoRestante is Presupuesto - CostoTramo,
        puede_viajar(Escala, Destino, PresupuestoRestante).

def campeon(partidos: List[(String, String, String)]): String = {
    val ganadores: List[String] = partidos.map {
        case(equipo1, _, "Equipo1") => equipo1
        case(_, equipo2, "Equipo2") => equipo2
    }

    val tablaPosiciones: Map[String, Int] = ganadores.groupBy(identity).map {
        case(equipo, victorias) => (equipo, victorias.length)
    }

    val equipoCampeon = tablaPosiciones.maxBy {
        case(equipo, puntos) => puntos
    }

    equipoCampeon._1
}
