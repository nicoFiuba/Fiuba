potencia(Base, Exp, Acumulador, Resultado) :- 
    Exp > 0,
    Exp1 is Exp - 1,
    NuevoAcumulador is Acumulador * Base,
    potencia(Base, Exp1, NuevoAcumulador, Resultado).

def esSolvente(consumos: List[List[(String, String, Int)]], tiposCambio: Map[String, Int], saldo: Int): Boolean = {
    
    val gastoTotal = consumos.flatten.filter{
        case(_, _, valor) => valor >= 0
    }.map {
        case(_, moneda, valor) => valor * tiposCambio(moneda)
        }.sum

    saldo >= gastoTotal
}
