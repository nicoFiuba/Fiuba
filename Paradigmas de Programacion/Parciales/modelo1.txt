/* Ejemplo

    val consumos1 = [[(“Rappi”, “Peso”, 7700)], [(“Jumbo”, “Peso”, 10300)]] // Suman 18000 pesos
    val consumos2 = [[(“Steam”, “Dolar”, 20)], [(“KLM”, “Euro”, 100), (“Apple”, “Euro”, -70)]] // Suman 170000 pesos

    val tiposCambio = { “Peso”: 1, “Dolar”: 1000, “Euro”: 1500 }
    
    esSolvente(consumos1, tiposCambio, 10000) ⇒ False
    esSolvente(consumos2, tiposCambio, 5000000) ⇒ True

*/

def esSolvente(consumos: List[List[(String, String, Int)]], tiposCambio: Map[String, Int], saldo: Int): Boolean = {
    
    val gastoTotal = consumos.flatten
    
    /* Consumos1
        
        List(("Rappi", "Peso", 7700),
            ("Jumbo", "Peso", 10300),
        )

    */

    /* Consumos2
        
        List(("Steam", "Dolar", 20),
            ("KLM", "Euro", 100),
            ("Apple", "Euro", -70)
        )

    */
    
    .filter{
        case(_, _, valor) => valor >= 0
    
        /* Consumos1
            
            List(("Rappi", "Peso", 7700),
                ("Jumbo", "Peso", 10300)
            )

        */

        /* Consumos2
            
            List(("Steam", "Dolar", 20),
                ("KLM", "Euro", 100)
            )

        */

    }
    .map {
        case(_, moneda, valor) => valor * tiposCambio(moneda)

        /* Consumos1
            
            List(1 * 7700, 1 * 10300) = List(7700, 10300)

        */

        /* Consumos2
            
            List(20 * 1000, 100 * 1500) = List(20000, 150000)

        */

    }
    
    .sum

    /* Consumos1
    
        7700 + 10300 = 18000
        
    */

    /* Consumos2
    
        20000 + 150000 = 170000
    
    */


    saldo >= gastoTotal

    /* Consumos1
    
        10000 >= 18000 => False
        
    */

    /* Consumos2
    
        5000000 >= 170000 => True
    
    */
}

FV((λx.((λy.(y x))(λz.(z w)))) a)
= (FV(λx.((λy.(y x))(λz.(z w)))) U FV(a))
= ((FV((λy.(y x))(λz.(z w))) - {x}) U {a})
= (((FV(λy.(y x)) U FV(λz.(z w))) - {x}) U {a})
= ((((FV(y x) - {y}) U (FV(z w) - {z})) - {x}) U {a})
= (((((FV(y) U FV(x)) - {y}) U ((FV(z) U FV(w)) - {z})) - {x}) U {a})
= ((((({y} U {x}) - {y}) U (({z} U {w}) - {z})) - {x}) U {a})
= (((({y, x} - {y}) U ({z, w} - {z})) - {x}) U {a})
= ((({x} U {w}) - {x}) U {a})
= (({x, w} - {x}) U {a})
= ({w} U {a})
= {w, a}

Por lo tanto las variables libres son 'w' y 'a' mientras que las ligadas son 'x', 'y', 'z' por λx, λy y λz
