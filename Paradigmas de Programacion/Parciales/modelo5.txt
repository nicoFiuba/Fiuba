/* Ejemplo
    
    apariciones(“Esto es una esto frase con frase con repeticiones frase”, 4) => 17

*/

def apariciones(frase: String, n: Int): Int = {
    frase.split(" ").toList

    /*
        List(“Esto", "es", "una", "esto", "frase", "con", "frase", "con", "repeticiones", "frase”)
    */
    
    .filter(palabra => palabra.length > n)
    
    /*
        List("frase", "frase",  "repeticiones", "frase")
    */

    .foldLeft(List.empty[String]) {
        case(unicas, palabra) if (unicas.contains(palabra)) => unicas
        case(unicas, palabra) => palabra :: unicas

        /*
            List("repeticiones", "frase")
        */

    }

    .foldLeft(0) {
        (total, palabra) => total + palabra.length
    }

    /*
        total = 0,
        0 + "repeticiones".length = 0 + 12 = 12
        12 + "frase".length = 12 + 5 = 17
    */

}
