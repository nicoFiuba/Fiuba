def mejoresVendedores(ventas: List[(String, Int)]): List[String] = {

    /* Ejemplo
        
        List(("Lucía", 1000),
            ("Pedro", 500),
            ("Lucía", 2000),
            ("Sofía", 500),
            ("Pedro", 1500),
            ("Lucía", 500),
            ("Mario", 2100)
        )

    */
    
    ventas.groupBy(_._1)
    /* 
        Map("Lucía" -> List(("Lucia", 1000), ("Lucia", 2000), ("Lucia", 500)),
            "Pedro" -> List(("Pedro", 500), ("Pedro", 1500)),
            "Sofia" -> List(("Sofia", 500)),
            "Mario" -> List(("Mario", 2100))
        )
    */
    
    .map { 
        case(nombre, vendido) => (nombre, vendido.map(_._2).sum)
    } // Map("Lucia" -> 3500, "Pedro" -> 2000, "Sofia" -> 500, "Mario" -> 2100)
    
    .filter {
        case(_, total) => total >= 2000
    } // Map("Lucia" -> 3500, "Pedro" -> 2000, "Mario" -> 2100)
    
    .toList // List(("Lucia", 3500), ("Pedro", 2000), ("Mario", 2100))
    
    .sortBy {
        case(_, total) => -total
    } // List(("Lucia", 3500), ("Mario", 2100), ("Pedro", 2000))
    
    .map {
        case(nombre, _) => nombre
    } // List(Lucia, Mario, Pedro)
}
