def reporteEquipos(participantes: List[(String, String, Int, List[String])]): Map[String, Int] = {

    participantes
        
        // Nos quedamos con los eventos "10k_1_Marzo" y "21k_6_Junio"
        .filter(participante => participante._2 == "10k_1_Marzo" || participante._2 == "21k_6_Junio")

        // Creamos una tupla (Equipo, Tiempo) para cada participante
        .flatMap(participante => participante._4.map(equipo => (equipo, participante._3)))
        
        // Agrupamos por equipo
        .groupBy(registro => registro._1)
        
        // Sumamos los tiempos de cada equipo
        .map {
            case (equipo, listaTiempos) => equipo -> listaTiempos.map(registro => registro._2).sum
        }
}
