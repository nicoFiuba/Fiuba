def gastoPorMarca(compras: List[(String, String, List[(String, Int)])], marcaBuscada: String): Map[String, Int] = {

    compras.foldLeft(Map.empty[String, Int]) {
        case (mapaAcumulado, (nombre, _, perfumes)) => val gastoPerfume = perfumes.foldLeft(0) {
            case(totalPerfumes, (perfume, precio)) if (perfume == marcaBuscada) => totalPerfumes + precio
            case(totalPerfumes, _) => totalPerfumes
        }

        val gastoAnterior = mapaAcumulado.getOrElse(nombre, 0)
        mapaAcumulado + (nombre -> (gastoAnterior + gastoPerfume))

        /*
            --- TRAZA DE MEMORIA ("Natura") ---
            
            Estado Inicial: Map()
            Vuelta 1 (Nico):
                gastoPerfume = 40
                mapaAcumulado = Map("Nico" -> 40)
                    
            Vuelta 2 (Matias):
                gastoPerfume = 35 
                mapaAcumulado = Map("Nico" -> 40, "Matias" -> 35)
                    
            Vuelta 3 (Lucas):
                gastoPerfume = 50
                mapaAcumulado = Map("Nico" -> 40, "Matias" -> 35, "Lucas" -> 50)
                    
            Vuelta 4 (Tomas):
                gastoPerfume = 0 (Solo compró Hugo Boss)
                mapaAcumulado = Map("Nico" -> 40, "Matias" -> 35, "Lucas" -> 50, "Tomas" -> 0)

        */

    }
}
