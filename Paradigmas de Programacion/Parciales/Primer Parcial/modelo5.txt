def apariciones(frase: String, n: Int): Int = {

    frase.split(" ").toList // List(“Esto", "es", "una", "esto", "frase", "con", "frase", "con", "repeticiones", "frase”)

    .filter(palabra => palabra.length > n) // List("frase", "frase",  "repeticiones", "frase")

    .distinct // List("frase", "repeticiones")

    .map(palabra => palabra.length) //Map("frase" -> 5, "repeticiones" -> 12)

    .sum // 5 + 12 = 17

}
