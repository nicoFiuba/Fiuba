% Reglas:
    % Caso Base
    puede_viajar(Origen, Destino, Presupuesto) :-
        viaje(Origen, Destino, Costo),
        Costo =< Presupuesto.

    % Recursion
    puede_viajar(Origen, Destino, Presupuesto) :-
        viaje(Origen, Escala, CostoTramo),
        CostoTramo =< Presupuesto,
        PresupuestoRestante is Presupuesto - CostoTramo,
        puede_viajar(Escala, Destino, PresupuestoRestante).
